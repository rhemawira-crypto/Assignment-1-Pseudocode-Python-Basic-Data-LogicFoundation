//filter - Produce a list of all orders with amount >= 50, su=ignificant or unsignificant.
FUNCTION filterProductByAmount(orders, threshold=50):   //threshold set to default 50
    
    SET significantResults = []      //variable to contain the significant order
    SET unsignificantResults = []    //variable to contain the unsignificant order (just in case if there is data that nee improvement)
    
    //add value into the right variable filter
    FOR EACH order IN orders:        
        IF order.amount >= threshold THEN
            ADD order TO significantResults
        ELSE
            ADD order TO unsignificantResults 
        END IF
    END FOR

    RETURN {   //return 2 value that contein list of significant and unsignificant orders
        significantOrders: significantResults,
        unsignificantOrders: unsignificantResults
        }
END FUNCTION

//revenue summary - aggregate (total, average amount, highest and lowest)
FUNCTION getDataAggregateOrders(orders):
    SET n = LENGTH(orders)

    //sum total revenue 
    SET total = 0
    FOR EACH order IN orders:
        SET total = total + order.amount
    END FOR
        
    //calculate average of order amount
    SET average = total / n
    SET average = ROUND(average, 2)    
    
    //search max and min value
    SET maxValue = orders[0].amount
    SET maxProduct = null
    SET minValue = orders[0].amount
    SET maxProduct = null
    FOR EACH order IN orders:
        IF order.amount > maxValue THEN
            SET maxValue = order.amount
            SET maxProduct = order.product
        END IF
    END FOR
    FOR EACH order IN orders:
        IF order.amount < minValue THEN
            SET minValue = order.amount
            SET minProduct = order.product
        END IF
    END FOR
    
    RETURN {            //return 6 value : total, average, highest amount - product, lowest amount - product
        totalRevenue: total,
        averagerOrderAmount: average,
        highestOrderAmount: maxValue,
        lowestOrderAmount: minValue,
        productHighestOrders: maxProduct,
        productLowestOrders: minProduct
        }
END FUNCTION

//group & count - per category
FUNCTION getDataGroupByCategory(orders):
    SET groups = {}
    SET counts = {}
    
    //iteration for grouping by category and counting
    FOR EACH order IN orders:
        IF NOT order.category IN groups THEN
            SET groups[order.category] = []   
            SET counts[order.category] = 1   
        ELSE
            ADD order TO groups[order.category]
            SET counts[order.category] = counts[order.category] + 1   
        END IF
    END FOR
    
    //iteration for summary of total revenue each category
    SET totalRevenueEachCategory = {}
    FOR EACH item IN groups:
        SET total = 0
        SET group = groups[item]
        FOR i FROM 0 TO LENGTH(group)-1:
            SET total = total + group[i].amount
        END FOR
        SET totalRevenueEachCategory[item] = total
    END FOR
      
    RETURN {        //return 3 results contain groups, counts, and total Revenue each category
            groups: groups,
            counts: counts,
            total: totalRevenueEachCategory
            }
END FUNCTION

//function for customer lookup - search
FUNCTION findOrdersByCustomer(orders, name):
    SET results = []
    SET target = LOWER(name)
    
    //search target
    FOR EACH order IN orders:
        IF target == LOWER(order.customer) THEN
            ADD order TO results
        END IF
    END FOR
    
    //print result in case user need to just search data customers, so it just can use this function
    IF LENGTH(results) == 0 THEN
        PRINT "============ DATA CUSTOMER ============"
        PRINT "-------------- Not Found --------------"
    ELSE
        PRINT "============ DATA CUSTOMER ============"
        PRINT "------------- ORDER FOUND -------------"
        PRINT ""
        PRINT "------ Here yours data for " + UPPER(target) + " ------"
        SET total = 0
        FOR EACH item IN results:
            SET total = total + item.amount
            PRINT "- Order: " + item.product + " (" + item.category + ") -> Amount : " + item.amount
        END FOR
        PRINT "" 
        PRINT "------ TOTAL ORDERS AMOUNT : " + total + " ------"
    END IF
    RETURN
END FUNCTION

//selection sorting, descending is default
FUNCTION selectionSorting(orders, descending = true)
    SET n = LENGTH(orders)
    FOR i FROM 0 TO n-2:
        SET targetIndex = i
        FOR j FROM i+1 TO n-1:
            IF descending THEN
                IF orders[j].amount > orders[targetIndex].amount THEN
                    SET targetIndex = j
                END IF
            ELSE
                IF orders[j].amount < orders[targetIndex].amount THEN
                    SET targetIndex = j
                END IF
            END IF
        END FOR
        IF targetIndex != i THEN
            SET temp = orders[i]
            SET orders[i] = orders[targetIndex]
            SET orders[targetIndex] = temp
        END IF
    END FOR
    RETURN orders
END FUNCTION

//Top customers — sort. Include total amount spent by each customers, sort from highest t lowest
FUNCTION getTopCustomers(orders):
    //grouping and count the total amount spent by each customers
    SET groups = {}
    FOR EACH order IN orders:
        SET nameCustomer = order.customer
        IF nameCustomer EXISTS IN groups THEN
            SET groups[nameCustomer] = groups[nameCustomer] + order.amount
        ELSE
            SET groups[nameCustomer] = order.amount
        END IF
    END FOR
    
    //turn from dictionary into list
    SET listCustomers = []
    FOR EACH item IN groups:
        SET amount = groups[item]
        ADD {customer: item, amount: amount} TO listCustomers
    END FOR
    
    //sorting the list with selection-sorting's function
    SET sortedListCustomers = selectionSorting(listCustomers)
    RETURN sortedListCustomers
END FUNCTION
