# filter - Produce a list of all orders with amount >= 50, significant or unsignificant.
def filter_product_by_amount(orders, threshold=50):
    significant_results = []
    unsignificant_results = []

    for order in orders:
        if order["amount"] >= threshold:
            significant_results.append(order)
        else:
            unsignificant_results.append(order)

    return {
        "significantOrders": significant_results,
        "unsignificantOrders": unsignificant_results,
    }


# revenue summary - aggregate (total, average amount, highest and lowest)
def get_data_aggregate_orders(orders):
    n = len(orders)

    # sum total revenue
    total = 0
    for order in orders:
        total = total + order["amount"]

    # calculate average of order amount
    average = total / n
    average = round(average, 2)

    # search max and min value
    max_value = orders[0]["amount"]
    max_product = None
    min_value = orders[0]["amount"]
    min_product = None

    for order in orders:
        if order["amount"] > max_value:
            max_value = order["amount"]
            max_product = order["product"]

    for order in orders:
        if order["amount"] < min_value:
            min_value = order["amount"]
            min_product = order["product"]

    return {
        "totalRevenue": total,
        "averagerOrderAmount": average,
        "highestOrderAmount": max_value,
        "lowestOrderAmount": min_value,
        "productHighestOrders": max_product,
        "productLowestOrders": min_product,
    }


# group & count - per category
def get_data_group_by_category(orders):
    groups = {}
    counts = {}

    # iteration for grouping by category and counting
    for order in orders:
        if order["category"] not in groups:
            groups[order["category"]] = []
            counts[order["category"]] = 1
        else:
            groups[order["category"]].append(order)
            counts[order["category"]] = counts[order["category"]] + 1

    # iteration for summary of total revenue each category
    total_revenue_each_category = {}
    for item in groups:
        total = 0
        group = groups[item]
        for i in range(0, len(group)):
            total = total + group[i]["amount"]
        total_revenue_each_category[item] = total

    return {
        "groups": groups,
        "counts": counts,
        "total": total_revenue_each_category,
    }


# function for customer lookup - search
def find_orders_by_customer(orders, name):
    results = []
    target = name.lower()

    # search target
    for order in orders:
        if target == order["customer"].lower():
            results.append(order)

    # print result
    if len(results) == 0:
        print("============ DATA CUSTOMER ============")
        print("-------------- Not Found --------------")
    else:
        print("============ DATA CUSTOMER ============")
        print("------------- ORDER FOUND -------------")
        print("")
        print("------ Here yours data for " + target.upper() + " ------")
        total = 0
        for item in results:
            total = total + item["amount"]
            print("- Order: " + item["product"] + " (" + item["category"] + ") -> Amount : " + str(item["amount"]))
        print("")
        print("------ TOTAL ORDERS AMOUNT : " + str(total) + " ------")
    return


# selection sorting, descending is default
def selection_sorting(orders, descending=True):
    n = len(orders)
    for i in range(0, n - 1):
        target_index = i
        for j in range(i + 1, n):
            if descending:
                if orders[j]["amount"] > orders[target_index]["amount"]:
                    target_index = j
            else:
                if orders[j]["amount"] < orders[target_index]["amount"]:
                    target_index = j
        if target_index != i:
            temp = orders[i]
            orders[i] = orders[target_index]
            orders[target_index] = temp
    return orders


# Top customers — sort. Include total amount spent by each customers, sort from highest to lowest
def get_top_customers(orders):
    # grouping and count the total amount spent by each customers
    groups = {}
    for order in orders:
        name_customer = order["customer"]
        if name_customer in groups:
            groups[name_customer] = groups[name_customer] + order["amount"]
        else:
            groups[name_customer] = order["amount"]

    # turn from dictionary into list
    list_customers = []
    for item in groups:
        amount = groups[item]
        list_customers.append({"customer": item, "amount": amount})

    # sorting the list with selection-sorting's function
    sorted_list_customers = selection_sorting(list_customers)
    return sorted_list_customers
